# PR Review Agent — Admin UI Implementation Plan (Plan 8 of N)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A bundled, server-rendered admin UI — single-admin login, a dashboard of queue stats + recent jobs + dead-letter, repo/config visibility, and a re-run action — backed by a small JSON API. Everything ships inside the same container (no separate Node build), guarded by session auth, and stays fully offline-testable.

**Architecture:** Two `APIRouter`s built from `(settings, queue)`: `app/api` returns JSON (stats, jobs, dead-letter, repos, re-run), `app/ui` renders Jinja pages (login, dashboard) with HTMX + Tailwind via CDN. `app/auth` provides stdlib password hashing (PBKDF2) and HMAC-signed session cookies — no new crypto dependency. Both routers are mounted into the existing `create_app` only when `ui.enabled`; webhook/health routes are unaffected and remain unauthenticated. Re-run resets a job row to `pending` via a new `Queue.requeue`, so the worker picks it up again.

**Tech Stack:** Python 3.11+, FastAPI, Jinja2, python-multipart (form posts), stdlib `hashlib`/`hmac`. Tailwind + HTMX are CDN `<script>`/`<link>` tags in templates (admin page, not a CSP-locked artifact). Existing: SQLAlchemy, httpx.

## Global Constraints

- Python 3.11+ (`from __future__ import annotations`).
- **No network in tests.** UI/API tested via FastAPI `TestClient`; auth helpers are pure functions.
- No new crypto dependency: passwords hashed with `hashlib.pbkdf2_hmac` (sha256), sessions signed with `hmac` + a configured `session_secret`.
- The UI is optional: mounted only when `settings.ui.enabled`. Webhook and `/health` routes stay public and unchanged.
- All `/admin/*` (except the login page/post) and all `/api/*` routes require a valid session cookie; unauthenticated requests get 401 (API) or redirect to login (UI pages).
- Secrets (password hash, session secret, tokens) come from config/env; never rendered into pages or returned by the API.
- `Queue.requeue(job_id)` resets an existing job to `status="pending", attempts=0` (re-run bypasses dedup by reusing the row, not inserting a duplicate).
- Every code change lands via TDD: failing test first, minimal impl, green, commit.

from __future__ import annotations

import base64
import hashlib
import hmac
import os


def hash_password(pw: str, *, salt: bytes | None = None, iterations: int = 200_000) -> str:
    salt = salt if salt is not None else os.urandom(16)
    dk = hashlib.pbkdf2_hmac("sha256", pw.encode(), salt, iterations)
    return (f"pbkdf2_sha256${iterations}${base64.b64encode(salt).decode()}"
            f"${base64.b64encode(dk).decode()}")


def verify_password(pw: str, stored: str) -> bool:
    try:
        algo, iters, salt_b64, hash_b64 = stored.split("$")
        if algo != "pbkdf2_sha256":
            return False
        salt = base64.b64decode(salt_b64)
        dk = hashlib.pbkdf2_hmac("sha256", pw.encode(), salt, int(iters))
        return hmac.compare_digest(base64.b64encode(dk).decode(), hash_b64)
    except Exception:
        return False


def make_session_token(user: str, secret: str) -> str:
    sig = hmac.new(secret.encode(), user.encode(), hashlib.sha256).hexdigest()
    return f"{user}:{sig}"


def verify_session_token(token: str | None, secret: str) -> str | None:
    if not token or ":" not in token:
        return None
    user, sig = token.rsplit(":", 1)
    expected = hmac.new(secret.encode(), user.encode(), hashlib.sha256).hexdigest()
    return user if hmac.compare_digest(expected, sig) else None
```

    def stats(self) -> dict:
        out = {"pending": 0, "in_progress": 0, "done": 0, "failed": 0}
        with self.session_factory() as s:
            rows = s.execute(select(Job.status, func.count()).group_by(Job.status)).all()
        for status, count in rows:
            out[status] = count
        return out

    def _job_dict(self, j) -> dict:
        return {"id": j.id, "platform": j.platform, "repo": j.repo,
                "pr_number": j.pr_number, "head_sha": j.head_sha, "task": j.task,
                "status": j.status, "attempts": j.attempts, "error": j.error}

    def list_jobs(self, limit: int = 50) -> list[dict]:
        with self.session_factory() as s:
            rows = s.execute(select(Job).order_by(Job.id.desc()).limit(limit)).scalars().all()
            return [self._job_dict(j) for j in rows]

    def list_dead_letter(self, limit: int = 50) -> list[dict]:
        with self.session_factory() as s:
            rows = s.execute(
                select(DeadLetter).order_by(DeadLetter.id.desc()).limit(limit)
            ).scalars().all()
            return [{"id": d.id, "job_id": d.job_id, "last_error": d.last_error} for d in rows]

    def requeue(self, job_id: int) -> bool:
        with self.session_factory() as s:
            job = s.get(Job, job_id)
            if job is None:
                return False
            job.status = "pending"
            job.attempts = 0
            job.error = None
            s.commit()
            return True
```

from __future__ import annotations

from fastapi import APIRouter, HTTPException, Request

from app.auth import verify_session_token


def build_api_router(settings, queue) -> APIRouter:
    router = APIRouter(prefix="/api")

    def _require(request: Request) -> str:
        user = verify_session_token(request.cookies.get("session"),
                                    settings.ui.session_secret)
        if not user:
            raise HTTPException(status_code=401)
        return user

    @router.get("/stats")
    def stats(request: Request):
        _require(request)
        return queue.stats()

    @router.get("/jobs")
    def jobs(request: Request):
        _require(request)
        return {"jobs": queue.list_jobs()}

    @router.get("/dead-letter")
    def dead_letter(request: Request):
        _require(request)
        return {"dead_letter": queue.list_dead_letter()}

    @router.get("/repos")
    def repos(request: Request):
        _require(request)
        return {"repos": [{"platform": r.platform.value, "full_name": r.full_name,
                           "tasks": r.tasks} for r in settings.repos]}

    @router.post("/jobs/{job_id}/rerun")
    def rerun(request: Request, job_id: int):
        _require(request)
        ok = queue.requeue(job_id)
        if not ok:
            raise HTTPException(status_code=404)
        return {"requeued": True}

    return router
```

        return verify_session_token(request.cookies.get("session"), secret)

    @router.get("/login", response_class=HTMLResponse)
    def login_page(request: Request):
        return _TEMPLATES.TemplateResponse(request, "login.html", {})

    @router.post("/login")
    def login(request: Request, username: str = Form(...), password: str = Form(...)):
        ok = (username == settings.ui.admin_user
              and verify_password(password, settings.ui.admin_password_hash))
        if not ok:
            return _TEMPLATES.TemplateResponse(
                request, "login.html", {"error": "Invalid credentials"}, status_code=401)
        resp = RedirectResponse("/admin", status_code=303)
        resp.set_cookie("session", make_session_token(username, secret),
                        httponly=True, samesite="lax")
        return resp

    @router.get("/logout")
    def logout():
        resp = RedirectResponse("/admin/login", status_code=303)
        resp.delete_cookie("session")
        return resp

    @router.get("", response_class=HTMLResponse)
    def dashboard(request: Request):
        if not _user(request):
            return RedirectResponse("/admin/login", status_code=303)
        return _TEMPLATES.TemplateResponse(request, "dashboard.html",
            {"stats": queue.stats(), "jobs": queue.list_jobs()})

    return router
```
