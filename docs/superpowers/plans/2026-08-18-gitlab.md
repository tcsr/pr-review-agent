# PR Review Agent — GitLab Support Implementation Plan (Plan 10 of N)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the agent work on GitLab too — a `GitLabAdapter` (merge requests over REST v4) that is a drop-in `GitPlatformAdapter`, wired through the factory, plus GitLab webhook parsing and token verification behind a platform-dispatched ingress. Self-hosted GitLab is supported via a per-repo `base_url`.

**Architecture:** `GitLabAdapter` mirrors `GitHubAdapter` but maps GitLab's merge-request API (URL-encoded project path, `PRIVATE-TOKEN` auth, notes/discussions, approve endpoint) onto the same neutral types. The ingress webhook handler becomes platform-aware: it dispatches signature verification and event parsing by the `{platform}` path segment (GitHub HMAC + `X-Hub-Signature-256` / `X-GitHub-Event`; GitLab shared-token + `X-Gitlab-Token` / `X-Gitlab-Event`), then shares the existing enqueue/comment-routing logic. Everything is offline-tested with `httpx.MockTransport` and FastAPI `TestClient`.

**Tech Stack:** Python 3.11+ (stdlib `urllib.parse`, `hmac`), httpx, FastAPI (existing). No new dependency.

## Global Constraints

- Python 3.11+ (`from __future__ import annotations`).
- **No network in tests.** Adapter via `httpx.MockTransport`; webhooks via `TestClient`; parsers/verifiers pure.
- `GitLabAdapter` implements every abstract `GitPlatformAdapter` method (all 12 incl. `update_comment`) so it instantiates and drops into the reviewer/worker unchanged.
- Neutral types are the only cross-boundary types; GitLab's `iid` maps to `pr_number`, its "notes" map to `Comment`, merge-request `sha` to `head_sha`.
- GitHub behavior is unchanged: the ingress refactor must keep all existing GitHub webhook tests green.
- Self-hosted GitLab uses `RepoConfig.base_url` (API root, e.g. `https://gitlab.example.com/api/v4`); default is `https://gitlab.com/api/v4`.
- GitLab webhook auth is a plain shared-secret token in `X-Gitlab-Token` (constant-time compare), NOT an HMAC — verify accordingly.
- Known degradations (documented, acceptable v1): GitLab inline comments post as a note prefixed with `` `path:line` `` (native positioned discussions deferred); `assign_reviewers` posts a note suggesting reviewers (GitLab needs numeric user IDs for real assignment); `request_changes` posts a note (GitLab has no request-changes verb; `approve` uses the approve endpoint).
- Every code change lands via TDD: failing test first, minimal impl, green, commit.

from __future__ import annotations

import re
import urllib.parse

from app.git.base import GitPlatformAdapter
from app.http import build_client, raise_for_status
from app.models import PRMeta, Diff, FileChange, Comment, Verdict, Side, PR, Platform

_ISSUE_REF = re.compile(r"#\d+")


def _count_diff(diff_text: str) -> tuple[int, int]:
    adds = dels = 0
    for line in (diff_text or "").splitlines():
        if line.startswith("+") and not line.startswith("+++"):
            adds += 1
        elif line.startswith("-") and not line.startswith("---"):
            dels += 1
    return adds, dels


class GitLabAdapter(GitPlatformAdapter):
    def __init__(self, project_path: str, token: str, transport=None,
                 api_base: str = "https://gitlab.com/api/v4"):
        self.repo = project_path
        self._client = build_client(api_base, {"PRIVATE-TOKEN": token},
                                    transport=transport)

    def _enc(self, repo: str) -> str:
        return urllib.parse.quote(repo, safe="")

    def _mr(self, repo: str, iid: int) -> str:
        return f"/projects/{self._enc(repo)}/merge_requests/{iid}"

    def _get(self, path: str):
        resp = self._client.get(path)
        raise_for_status(resp)
        return resp.json()

    def _post(self, path: str, payload: dict):
        resp = self._client.post(path, json=payload)
        raise_for_status(resp)
        return resp.json()

    def get_pr_meta(self, repo, pr_number) -> PRMeta:
        d = self._get(self._mr(repo, pr_number))
        status = (d.get("head_pipeline") or {}).get("status")
        ci_passed = True if status == "success" else (
            False if status in ("failed", "canceled") else None)
        body = d.get("description") or ""
        return PRMeta(
            title=d.get("title") or "",
            body=body,
            author=d.get("author", {}).get("username", ""),
            base_sha=(d.get("diff_refs") or {}).get("base_sha", ""),
            head_sha=d.get("sha", ""),
            labels=list(d.get("labels", [])),
            linked_issues=_ISSUE_REF.findall(body),
            ci_passed=ci_passed,
        )

    def get_diff(self, repo, pr_number) -> Diff:
        d = self._get(self._mr(repo, pr_number) + "/changes")
        files = []
        for c in d.get("changes", []):
            adds, dels = _count_diff(c.get("diff", ""))
            status = ("added" if c.get("new_file") else
                      "deleted" if c.get("deleted_file") else "modified")
            files.append(FileChange(path=c["new_path"], patch=c.get("diff", ""),
                                    additions=adds, deletions=dels, status=status))
        return Diff(files=files)

    # write methods (Task 2)
    def post_comment(self, repo, pr_number, body) -> str:
        raise NotImplementedError

    def post_inline_comment(self, repo, pr_number, path, line, body, side: Side) -> str:
        raise NotImplementedError

    def set_review_verdict(self, repo, pr_number, verdict: Verdict, body) -> None:
        raise NotImplementedError

    def get_thread(self, repo, pr_number) -> list[Comment]:
        raise NotImplementedError

    def reply(self, repo, pr_number, in_reply_to, body) -> str:
        raise NotImplementedError

    def update_comment(self, repo, pr_number, comment_id, body) -> None:
        raise NotImplementedError

    def list_open_prs(self, repo) -> list[PR]:
        raise NotImplementedError

    def add_labels(self, repo, pr_number, labels) -> None:
        raise NotImplementedError

    def assign_reviewers(self, repo, pr_number, reviewers) -> None:
        raise NotImplementedError

    def post_comment(self, repo, pr_number, body) -> str:
        r = self._post(self._mr(repo, pr_number) + "/notes", {"body": body})
        return str(r["id"])

    def post_inline_comment(self, repo, pr_number, path, line, body, side: Side) -> str:
        r = self._post(self._mr(repo, pr_number) + "/notes",
                       {"body": f"`{path}:{line}` {body}"})
        return str(r["id"])

    def set_review_verdict(self, repo, pr_number, verdict: Verdict, body) -> None:
        if verdict == Verdict.approve:
            self._post(self._mr(repo, pr_number) + "/approve", {})
        else:
            self._post(self._mr(repo, pr_number) + "/notes", {"body": body})

    def get_thread(self, repo, pr_number) -> list[Comment]:
        rows = self._get(self._mr(repo, pr_number) + "/notes")
        return [Comment(id=str(c["id"]), author=c.get("author", {}).get("username", ""),
                        body=c.get("body", "")) for c in rows]

    def reply(self, repo, pr_number, in_reply_to, body) -> str:
        r = self._post(self._mr(repo, pr_number) + "/notes", {"body": body})
        return str(r["id"])

    def update_comment(self, repo, pr_number, comment_id, body) -> None:
        resp = self._client.put(self._mr(repo, pr_number) + f"/notes/{comment_id}",
                                json={"body": body})
        raise_for_status(resp)

    def list_open_prs(self, repo) -> list[PR]:
        rows = self._get(f"/projects/{self._enc(repo)}/merge_requests?state=opened")
        return [PR(platform=Platform.gitlab, repo=repo, number=r["iid"],
                   head_sha=r.get("sha", "")) for r in rows]

    def add_labels(self, repo, pr_number, labels) -> None:
        resp = self._client.put(self._mr(repo, pr_number),
                                json={"add_labels": ",".join(labels)})
        raise_for_status(resp)

    def assign_reviewers(self, repo, pr_number, reviewers) -> None:
        body = "Suggested reviewers: " + " ".join(f"@{r}" for r in reviewers)
        self._post(self._mr(repo, pr_number) + "/notes", {"body": body})
```

_GL_ACCEPTED = {"open", "reopen", "update"}


def parse_gitlab_event(payload: dict, event_type: str | None) -> Event | None:
    if event_type == "Merge Request Hook":
        attrs = payload.get("object_attributes", {})
        if attrs.get("action") not in _GL_ACCEPTED:
            return None
        return Event(
            platform=Platform.gitlab,
            repo=payload["project"]["path_with_namespace"],
            pr_number=attrs["iid"],
            head_sha=attrs["last_commit"]["id"],
            action=attrs["action"],
        )
    if event_type == "Note Hook":
        mr = payload.get("merge_request")
        if not mr:
            return None
        attrs = payload["object_attributes"]
        return Event(

from app.ingress.events import parse_github_event, parse_gitlab_event
from app.ingress.security import verify_github_signature, verify_gitlab_token
...
    @app.post("/webhook/{platform}")
    async def webhook(platform: str, request: Request):
        body = await request.body()
        secret = settings.ingress.webhook_secret
        if platform == "github":
            if verify and secret and not verify_github_signature(
                    secret, body, request.headers.get("X-Hub-Signature-256")):
                return Response(status_code=401)
            parse = lambda p: parse_github_event(p, request.headers.get("X-GitHub-Event"))
        elif platform == "gitlab":
            if verify and secret and not verify_gitlab_token(
                    secret, request.headers.get("X-Gitlab-Token")):
                return Response(status_code=401)
            parse = lambda p: parse_gitlab_event(p, request.headers.get("X-Gitlab-Event"))
        else:
            return Response(status_code=404)
        try:
            payload = json.loads(body)
        except json.JSONDecodeError:
            return Response(status_code=400)
        event = parse(payload)
        if event is None:
            return {"enqueued": 0}
        repo_cfg = find_repo(settings, event.platform.value, event.repo)
        if repo_cfg is None:
            return {"enqueued": 0}
        if event.action == "comment":
            handle = settings.bot_handle
            if handle and event.comment_author == handle:
                return {"enqueued": 0}
            if not handle or f"@{handle}".lower() not in (event.comment_body or "").lower():
                return {"enqueued": 0}
            jobs = queue.enqueue_event(event, ["qa_reply"])
            return {"enqueued": len(jobs)}
        jobs = queue.enqueue_event(event, resolve_tasks(settings, repo_cfg))
        return {"enqueued": len(jobs)}
```
